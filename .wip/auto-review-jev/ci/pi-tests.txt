  it('invalidates Pi cached and pending reviews on Jev key changes', async () => {
    let revision = 'key-1';
    let release!: (value: { verdict: 'allow' }) => void;
    const reviewer = vi.fn().mockResolvedValue({ verdict: 'allow' });
    const deps = buildDeps(reviewer);
    deps.getAutoReviewRuntimePolicy = () => ({ forceHost: true, revision });
    const handle = await new PiAgent(deps).startSession({ sessionId: 'jev-cache', workingDir: cwd, model: 'm', permissionMode: 'auto' });
    const action = { kind: 'exec' as const, command: 'pnpm test' };
    try {
      await handle.reviewAutoPermissionAction!(action);
      await handle.reviewAutoPermissionAction!(action);
      expect(reviewer).toHaveBeenCalledOnce();
      revision = 'key-2';
      reviewer.mockImplementationOnce(() => new Promise(resolve => { release = resolve; }));
      const pending = handle.reviewAutoPermissionAction!(action);
      await vi.waitFor(() => expect(reviewer).toHaveBeenCalledTimes(2));
      revision = 'key-3'; release({ verdict: 'allow' });
      expect((await pending).verdict).toBe('block');
      expect((await handle.reviewAutoPermissionAction!(action)).verdict).toBe('allow');
      expect(reviewer).toHaveBeenCalledTimes(3);
    } finally { await handle.close(); }
  });
